from scipy import stats

A = [32, 30, 28, 35, 33]
B = [40, 42, 38, 41, 39]
C = [36, 34, 37, 35, 38]

F, p = stats.f_oneway(A, B, C)
print(round(F, 3), round(p, 5))   # → 21.528 0.00011
